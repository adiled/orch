followed by railroad (syntax) diagrams for the key production rules.

\section{EBNF Grammar}
% ============================================================================

\subsection{Document Structure}

\begin{lstlisting}
/* Directives start at column 0. Trailing whitespace on a line is not part of any value. */
orchfile    ::= bom? line ( newline line )*   /* the final line ending is optional */
line        ::= blank | comment | ( version | arg | service_decl | directive ) ws?
blank       ::= ws*
comment     ::= ws* '#' any*
bom         ::= #xFEFF   /* optional UTF-8 byte-order mark; ignored */
\end{lstlisting}

\subsection{Top-level Declarations}

\begin{lstlisting}
version     ::= 'ORCH_VERSION' ws version_number
arg         ::= 'ARG' ws identifier '=' value
service_decl ::= 'SERVICE' ws service_name
\end{lstlisting}

\subsection{Directives}

\begin{lstlisting}
directive   ::= mode_directive
              | container_directive
              | host_directive
              | common_directive
              | clear_directive
\end{lstlisting}

\subsubsection{Overlay Composition}

\begin{lstlisting}
/* Overlay composition - resets a list-type directive before merge */
clear_directive
            ::= 'CLEAR' ws clearable_target

clearable_target
            ::= 'ENV' | 'ENV_FILE' | 'PUBLISH' | 'VOLUME'
              | 'REQUIRES' | 'AFTER'
\end{lstlisting}

\subsubsection{Execution Mode (mutually exclusive per service)}

\begin{lstlisting}
/* Execution mode - mutually exclusive per service */
mode_directive
            ::= 'FROM' ws image_ref
              | 'RUN' ws command_string
\end{lstlisting}

\subsubsection{Container-only Directives}

\begin{lstlisting}
/* Container-only */
container_directive
            ::= 'ENTRYPOINT' ws command_string
              | 'CMD' ws command_string
              | 'PUBLISH' ws ( host_address ':' )? port ':' port
              | 'VOLUME' ws volume_source ':' volume_destination ( ':' volume_mode )?
\end{lstlisting}

\subsubsection{Host-only Directives}

\begin{lstlisting}
/* Host-only */
host_directive
            ::= 'USER' ws user_name
              | 'STOP' ws command_string
              | 'RELOAD' ws command_string
\end{lstlisting}

\subsubsection{Common Directives}

\begin{lstlisting}
/* Common - environment */
env_directive
            ::= 'WORKDIR' ws path
              | 'ENV' ws env_key '=' value
              | 'ENV_FILE' ws path

/* Common - dependencies */
dependency_directive
            ::= 'REQUIRES' ws service_name ( ws service_name )*
              | 'AFTER' ws service_name ( ws service_name )*

/* Common - health */
health_directive
            ::= 'HEALTHCHECK' ws ( url | command_string )
              | 'READINESS_TIMEOUT' ws duration

/* Common - lifecycle */
lifecycle_directive
            ::= 'ONESHOT' ws bool
              | 'DISABLED' ws bool
              | 'RECREATE' ws recreate_policy

/* Common - restart */
restart_directive
            ::= 'RESTART' ws restart_policy
              | 'RESTART_DELAY' ws duration
              | 'START_LIMIT_BURST' ws integer
              | 'START_LIMIT_INTERVAL' ws duration

/* Common - timeouts */
timeout_directive
            ::= 'TIMEOUT_START' ws duration
              | 'TIMEOUT_STOP' ws duration

/* Common - resources */
resource_directive
            ::= 'MEMORY' ws memory_size
              | 'CPUS' ws number
              | 'CPU_QUOTA' ws percentage
              | 'LIMIT_NOFILE' ws integer
              | 'LIMIT_NPROC' ws integer
              | 'TASKS_MAX' ws integer
              | 'IO_WEIGHT' ws integer   /* 10-1000 */

/* Common - logging */
logging_directive
            ::= 'STDOUT' ws path
              | 'STDERR' ws path

common_directive
            ::= env_directive
              | dependency_directive
              | health_directive
              | lifecycle_directive
              | restart_directive
              | timeout_directive
              | resource_directive
              | logging_directive
\end{lstlisting}

\subsection{Terminals}

\begin{lstlisting}
/* Terminals */
service_name ::= letter ( letter | digit | '-' )*   /* max 63 chars */
identifier  ::= ( alpha | '_' ) ( alpha | digit | '_' )*   /* ARG names */
env_key     ::= identifier
user_name   ::= ( alpha | digit | '_' ) ( alpha | digit | '_' | '-' | '.' )*

volume_source      ::= volume_part   /* host path or named volume */
volume_destination ::= volume_part   /* container path */
volume_part ::= ( visible - ':' ) ( ( any - ':' )* ( visible - ':' ) )?
volume_mode ::= 'ro' | 'rw'          /* rw if omitted */

/* Free text. It never starts or ends with whitespace. */
text        ::= visible ( any* visible )?
image_ref   ::= text
command_string ::= text
path        ::= text
value       ::= ( any* visible )?   /* after '='; may be empty */
url         ::= ( 'http://' | 'https://' ) text

/* var_ref and var_escape may appear anywhere in the value of a directive inside a SERVICE block. Each value must match its rule after expansion (see SPEC.md). */
var_ref     ::= '${' identifier '}'   /* identifier MUST be a declared ARG */
var_escape  ::= '$${'                 /* a literal '${'; never starts a var_ref */

bool        ::= 'true' | 'false'
restart_policy ::= 'no' | 'always' | 'on-failure'
recreate_policy ::= 'always' | 'never'

duration    ::= integer ( 's' | 'm' )
memory_size ::= integer ( 'K' | 'M' | 'G' )
percentage  ::= integer '%'
number      ::= integer ( '.' digit+ )?
integer     ::= digit+
port        ::= digit+   /* 1-65535 */
version_number ::= integer '.' integer '.' integer ( '-' prerelease )?
prerelease  ::= ( letter | digit | '-' | '.' )+   /* e.g. rc, beta.1 */

host_address ::= ipv4 | ipv6 | hostname
ipv4        ::= octet '.' octet '.' octet '.' octet
octet       ::= digit+                          /* 0-255 */
ipv6        ::= ( hex_digit | ':' | '.' )+      /* RFC 4291 text form, without brackets */
hostname    ::= label ( '.' label )*            /* RFC 1123 */
label       ::= alnum ( ( alnum | '-' )* alnum )?   /* max 63 chars */
alnum       ::= alpha | digit
hex_digit   ::= digit | [a-fA-F]

letter      ::= [a-z]
alpha       ::= [a-zA-Z]
digit       ::= [0-9]
ws          ::= [ #x9]+
newline     ::= #xA | #xD #xA
any         ::= #x9 | [#x20-#x7E] | [#xA0-#xD7FF] | [#xE000-#x10FFFF]   /* no control characters except tab; no surrogates */
visible     ::= any - [ #x9]
\end{lstlisting}

\medskip
\noindent
\textbf{Note:} The grammar describes each value \emph{after} variable
expansion. \texttt{var\_ref} (\texttt{\$\{name\}}) may appear anywhere in the
value of a directive inside a SERVICE block, and \texttt{var\_escape}
(\texttt{\$\$\{}) stands for a literal \texttt{\$\{}. With
\texttt{ARG port=8080}, \texttt{PUBLISH \$\{port\}:80} is checked as
\texttt{PUBLISH 8080:80}. Every name must be a declared ARG (C5).

\section{Constraints}
% ============================================================================

The following constraints are enforced by the parser and cannot be expressed
in the EBNF grammar alone:

\begin{description}
  \item[C1: FROM XOR RUN] A service must specify exactly one of \texttt{FROM}
    (container mode) or \texttt{RUN} (host mode). Both or neither is an error.

  \item[C2: Container-only directives] \texttt{ENTRYPOINT}, \texttt{CMD},
    \texttt{PUBLISH}, \texttt{VOLUME} are only valid with \texttt{FROM}.

  \item[C3: Host-only directives] \texttt{USER}, \texttt{STOP},
    \texttt{RELOAD} are only valid with \texttt{RUN}.

  \item[C4: Dependency acyclicity] \texttt{REQUIRES} and \texttt{AFTER}
    edges must form a directed acyclic graph (DAG). Cycles are a parse error.

  \item[C5: Declared variables] Every \texttt{\$\{name\}} must name an
    \texttt{ARG} declared in one of the merged files.

  \item[C6: One entry per key] Within a service, each container port is
    published at most once (\texttt{PUBLISH}) and each destination is mounted
    at most once (\texttt{VOLUME}).

  \item[C7: No repeats within a file] Within one file, a service sets each
    scalar directive and each \texttt{ENV} key at most once. Across files,
    overlays replace values.
\end{description}

\noindent Numeric ranges noted in the grammar's comments (ports 1--65535, IPv4
octets 0--255, hostname labels and service names up to 63 characters,
\texttt{IO\_WEIGHT} 10--1000) are also enforced by the parser.
